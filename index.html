/* LÓGICA DE PROCESAMIENTO CORREGIDA BAJO TU REGLA ESTRICTA */ function processImageOCR(event) { const file = event.target.files[0]; if (!file) return; const ocrIcon = document.getElementById('btn-ocr-icon'); const ocrText = document.getElementById('btn-ocr-text'); ocrIcon.className = "fa-solid fa-spinner animate-spin text-amber-400"; ocrText.innerText = "LEYENDO..."; showToast("Procesando reporte POS..."); // REGLA OBLIGATORIA: Dejar absolutamente todo el plano en blanco antes de evaluar tables.forEach(t => t.status = 'free'); const reader = new FileReader(); reader.onload = function(e) { const img = new Image(); img.onload = function() { const canvas = document.createElement('canvas'); const ctx = canvas.getContext('2d'); canvas.width = img.width; canvas.height = img.height; ctx.drawImage(img, 0, 0); // Optimización de binarización para mejorar la lectura del espacio y la barra '/' const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height); const data = imgData.data; for (let i = 0; i < data.length; i += 4) { const brightness = 0.34 * data[i] + 0.5 * data[i + 1] + 0.16 * data[i + 2]; const threshold = brightness < 150 ? 0 : 255; data[i] = data[i+1] = data[i+2] = threshold; } ctx.putImageData(imgData, 0, 0); Tesseract.recognize( canvas.toDataURL(), 'eng', { logger: m => console.log(m) } ).then(({ data: { text } }) => { // Normalizamos el texto base manteniendo los espacios limpios const cleanText = text.toLowerCase().replace(/\s+/g, ' '); // EXPRESIÓN REGULAR EXACTA: // tbl -> Busca la palabra fija // \s -> Obliga a que exista el espacio en blanco posterior // (\d+)-> Captura el número de la mesa // (?=\/) -> Mira hacia adelante para asegurarse de que lo que sigue es una barra '/' const regex = /tbl\s(\d+)(?=\/)/g; let match; const detectedTables = []; while ((match = regex.exec(cleanText)) !== null) { detectedTables.push(match[1]); // Extrae con precisión el número capturado } // Mapeo directo sobre el plano interactivo let matchedCount = 0; tables.forEach(t => { if (detectedTables.includes(t.label)) { t.status = 'occupied'; // Pinta la mesa encontrada en negro matchedCount++; } }); if (matchedCount > 0) { showToast(`Escaneo exitoso: ${matchedCount} mesas detectadas.`); } else { showToast("No se detectaron mesas con el formato 'Tbl [número]/'."); } saveData(); render(); }).catch(err => { console.error(err); showToast("Error en el módulo de lectura."); render(); }).finally(() => { ocrIcon.className = "fa-solid fa-camera text-sm"; ocrText.innerText = "SCAN"; document.getElementById('ocr-camera-input').value = ""; }); }; img.src = e.target.result; }; reader.readAsDataURL(file); }
